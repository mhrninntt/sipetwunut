// Muat Data Pola Ruang fetch('https://mhrninntt.github.io/sipettulung/polaruangwunutbaru.geojson') .then(res => res.json()) .then(data => { polaRuangLayer = L.geoJSON(data, { style: stylePolaRuang, onEachFeature: function(f, l) { // Membuat desain popup tabel untuk Pola Ruang let popupContent = `
Informasi Pola Ruang
Kode : ${f.properties.KODE || '-'}
Zona : ${f.properties.ZONA || '-'}
Sub Zona : ${f.properties.SUB_ZONA || '-'}
`; l.bindPopup(popupContent); l.on('mouseover', function(e) { e.target.setStyle({fillOpacity: 1, weight: 3, color: '#000'}); }); l.on('mouseout', function(e) { polaRuangLayer.resetStyle(e.target); }); } }); if (document.getElementById('check-polaruang').checked) { map.addLayer(polaRuangLayer); } renderLegend(); }) .catch(e => console.error("Gagal Muat Data Pola Ruang:", e));
